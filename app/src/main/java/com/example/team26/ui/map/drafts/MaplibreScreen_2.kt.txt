// When testing on emulator, set useTestLocation to true and provide your test coordinates
val useTestLocation = true
val testLatitude = 64.136181 // Gróska
val testLongitude = -21.946960

@Composable
fun MaplibreScreen(modifier: Modifier = Modifier, navController: NavHostController, stationViewModel: StationViewModel) {
    // Collect station data as state
    val allStations by stationViewModel.allStationsUiState.collectAsState()
    val selectedStationId by stationViewModel.selectedStationId.collectAsState()

    // Default position (Reykjavik) as fallback
    val defaultPosition = Position(-21.9408, 64.147)

    // Log stations for debugging
    LaunchedEffect(allStations) {
        Log.d("MaplibreScreen", "Number of stations: ${allStations.size}")
        allStations.forEach { station ->
            Log.d("MaplibreScreen", "Station: id=${station.stationId}, name=${station.stationName}, " +
                    "lat=${station.latitude}, lng=${station.longitude}, status=${station.status}")
        }
    }

    // State to hold user location
    var userPosition by remember { mutableStateOf(defaultPosition) }
    val context = LocalContext.current
    val fusedLocationClient = remember { LocationServices.getFusedLocationProviderClient(context) }

    val defaultZoom = 14.0
    // Create camera state
    val cameraPosition = CameraPosition(
        target = defaultPosition,
        zoom = defaultZoom,
        bearing = 0.0,
        tilt = 0.0,
        padding = PaddingValues(0.dp)
    )

    val camera = rememberCameraState(cameraPosition)
    val styleState = rememberStyleState()

    // Wrap map in the permission handler
    PermissionHandler(
        modifier = modifier,
        permission = Manifest.permission.ACCESS_FINE_LOCATION,
        description = "Location permission is needed to show your position on the map",
        onGranted = {
            // Update the camera when we get a new position
            LaunchedEffect(userPosition) {
                camera.animateTo(
                    CameraPosition(
                        target = userPosition,
                        zoom = defaultZoom
                    ),
                    duration = 700.milliseconds
                )
            }

            // Get location once permission is granted
            LaunchedEffect(Unit) {
                // Use the test location if testing on emulator
                if (useTestLocation) {
                    val mockLocation = Location("MockProvider").apply {
                        latitude = testLatitude
                        longitude = testLongitude
                    }
                    userPosition = Position(mockLocation.longitude, mockLocation.latitude)
                } else {
                    getLastLocation(fusedLocationClient) { location ->
                        userPosition = Position(location.longitude, location.latitude)
                    }
                }
            }

            // Render the map
            Box(Modifier.fillMaxSize()) {
                MaplibreMap(
                    cameraState = camera,
                    styleUri = "https://tiles.openfreemap.org/styles/liberty",
                    styleState = styleState,
                    ornamentSettings = OrnamentSettings(
                        isLogoEnabled = false,
                        isAttributionEnabled = true,
                        isCompassEnabled = true,
                        isScaleBarEnabled = true,
                    ),
                    gestureSettings = GestureSettings(
                        isTiltGesturesEnabled = true,
                        isZoomGesturesEnabled = true,
                        isRotateGesturesEnabled = true,
                        isScrollGesturesEnabled = true,
                    )
                ) {

                    val stationsSource = rememberGeoJsonSource(
                        id = "stations-source",
                        uri = "asset://stations.geojson"
                    )
                    allStations.forEach { station ->
                        // Use standard Kotlin conditionals to determine visual properties
                        val isSelected = station.stationId == selectedStationId
                        val radius = if (isSelected) 8.dp else 5.dp

                        // Simple when statement for color based on status
                        val color = when (station.status) {
                            "ON" -> Color(0xFF00CC00)  // Green
                            "OFF" -> Color(0xFFCC0000) // Red
                            else -> Color(0xFF888888)  // Gray
                        }

                        // Filter to show only this specific station
                        val filterExpression =
                            (Feature.get("stationId") as Expression<EquatableValue>)
                                .eq(const(station.stationId.toString()) as Expression<EquatableValue>)

                        // Circle layer with pre-computed values from native Kotlin
                        CircleLayer(
                            id = "circle-${station.stationId}",
                            source = stationsSource,
                            filter = filterExpression,
                            radius = const(radius),  // Pre-computed using Kotlin if/else
                            color = const(color),    // Pre-computed using Kotlin when
                            strokeWidth = const(1.dp),
                            strokeColor = const(Color.White),
                            onClick = {
                                Log.d("MaplibreScreen", "Clicked station: ${station.stationId}")
                                stationViewModel.setStationId(station.stationId)
                                ClickResult.Consume
                            }
                        )

                        // Text label layer for this station
                        SymbolLayer(
                            id = "label-${station.stationId}",
                            source = stationsSource,
                            filter = filterExpression,
                            textField = const(station.stationName),
                            textSize = const(1.em),
                            textOffset = dev.sargunv.maplibrecompose.expressions.dsl.offset(
                                0.em,
                                0.6.em
                            ),
                            textColor = const(Color.Black),
                            textHaloColor = const(Color.White),
                            textHaloWidth = const(1.dp)
                        )
                    }
                }

                    Box(modifier = Modifier.fillMaxSize().padding(8.dp)) {
                        DisappearingScaleBar(
                            metersPerDp = camera.metersPerDpAtTarget,
                            zoom = camera.position.zoom,
                            modifier = Modifier.align(Alignment.TopStart),
                        )
                        DisappearingCompassButton(
                            camera,
                            modifier = Modifier.align(Alignment.TopEnd)
                        )
                        AttributionButton(
                            styleState,
                            modifier = Modifier.align(Alignment.BottomEnd)
                        )
                    }

            }
        }
    )
}
